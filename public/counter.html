using System; // Streamer.bot -> Actions -> (new action) -> Sub-Actions -> Core -> C# -> "Execute C# Code" // Is poori file ko paste karo, "Save and Compile" dabao. // Ek hi script sab karta hai. Action ko argument "action" do (neeche list hai). // action = death+ | death- | win | loss | reset | refresh // Triggers: chat commands (!death+ !death- !win !loss !resetstats) ya Stream Deck/hotkey. // Chat commands ko Mods/Broadcaster tak restrict karna (Command settings me Permissions). public class CPHInline { // ---- SIRF YE 3 CHEEZEIN BADLO ---- const string SCENE = "Game"; // jis OBS scene me ye text sources rakhe hain const string DEATH_SRC = "DeathCount"; // OBS "Text (GDI+)" source ka naam const string WL_SRC = "WinLoss"; // OBS "Text (GDI+)" source ka naam // ---------------------------------- public bool Execute() { string action = args.ContainsKey("action") ? args["action"].ToString().ToLower() : "refresh"; int deaths = CPH.GetGlobalVar("sc_deaths", true); int wins = CPH.GetGlobalVar("sc_wins", true); int losses = CPH.GetGlobalVar("sc_losses", true); switch (action) { case "death+": deaths++; break; case "death-": if (deaths > 0) deaths--; break; case "win": wins++; break; case "loss": losses++; break; case "reset": deaths = 0; wins = 0; losses = 0; break; } CPH.SetGlobalVar("sc_deaths", deaths, true); CPH.SetGlobalVar("sc_wins", wins, true); CPH.SetGlobalVar("sc_losses", losses, true); CPH.ObsSetGdiText(SCENE, DEATH_SRC, "Deaths: " + deaths); CPH.ObsSetGdiText(SCENE, WL_SRC, "W " + wins + " - " + losses + " L"); // chat me current score (YouTube) if (action != "refresh") CPH.SendYouTubeMessage("Deaths: " + deaths + " | Ranked: " + wins + "W - " + losses + "L"); return true; } }
